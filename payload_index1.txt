    <script>
        const chatFeed = document.getElementById('chat-feed');
        const userInput = document.getElementById('user-input');
        const sendBtn = document.getElementById('send-btn');
        const brandCenter = document.getElementById('brand-center');
        const typingIndicator = document.getElementById('typing-pulse');
        const chatViewport = document.getElementById('chat-viewport');
        const clockDisp = document.getElementById('clock-disp');

        // Clock System
        setInterval(() => {
            const d = new Date();
            clockDisp.innerText = d.toLocaleTimeString('en-US', {hour12: false});
        }, 1000);

        let chatMemory = [];
        const MAX_MEMORY = 8;
        let isFirstMessage = true;

        function clearMemory() {
            chatMemory = [];
            chatFeed.innerHTML = '';
            brandCenter.classList.remove('hide');
            isFirstMessage = true;
            appendMessage("Cache memory flushed. Awaiting new instructions.", 'ai');
        }

        // --- RENDER TEXT DENGAN KODE (MONOCHROME) ---
        function formatMessage(text) {
            const codeBlockRegex = /```(\w+)?\n([\s\S]*?)```/g;
            let formattedText = text.replace(codeBlockRegex, (match, language, code) => {
                const langLabel = language ? language : 'CODE';
                // Escape HTML tags untuk mencegah XSS
                const escapedCode = code.replace(/</g, "&lt;").replace(/>/g, "&gt;");
                return `
                    <div class="code-wrapper">
                        <div class="code-header">
                            <span>${langLabel}</span>
                            <button class="copy-btn" onclick="copyCode(this)">
                                <svg viewBox="0 0 24 24"><rect x="9" y="9" width="13" height="13" rx="2" ry="2"></rect><path d="M5 15H4a2 2 0 0 1-2-2V4a2 2 0 0 1 2-2h9a2 2 0 0 1 2 2v1"></path></svg>
                                Copy
                            </button>
                        </div>
                        <pre><code>${escapedCode}</code></pre>
                    </div>
                `;
            });

            formattedText = formattedText.replace(/\*\*(.*?)\*\*/g, '<strong>$1</strong>');
            formattedText = formattedText.replace(/\*(.*?)\*/g, '<em>$1</em>');
            formattedText = formattedText.replace(/\n/g, '<br>');

            return formattedText;
        }

        // --- FUNGSI COPY KODE ---
        function copyCode(button) {
            const pre = button.parentElement.nextElementSibling;
            const codeText = pre.innerText;

            navigator.clipboard.writeText(codeText).then(() => {
                const originalText = button.innerHTML;
                button.innerHTML = '<svg viewBox="0 0 24 24"><polyline points="20 6 9 17 4 12"></polyline></svg> Copied';
                button.classList.add('copied');
                setTimeout(() => {
                    button.innerHTML = originalText;
                    button.classList.remove('copied');
                }, 2000);
            });
        }

        // --- TAMBAH PESAN KE UI ---
        function appendMessage(text, sender) {
            const row = document.createElement('div');
            row.className = `chat-row ${sender === 'user' ? 'user-row' : 'ai-row'}`;
            
            const bubble = document.createElement('div');
            bubble.className = 'message-bubble';
            
            bubble.innerHTML = sender === 'ai' ? formatMessage(text) : text.replace(/</g, "&lt;").replace(/>/g, "&gt;");

            row.appendChild(bubble);
            chatFeed.appendChild(row);
            
            // Auto Scroll
            setTimeout(() => {
                chatViewport.scrollTop = chatViewport.scrollHeight;
            }, 50);
        }

        // --- KIRIM PESAN KE API ---
        async function sendMessage() {
            const text = userInput.value.trim();
            if (!text) return;

            if (isFirstMessage) {
                brandCenter.classList.add('hide');
                isFirstMessage = false;
            }

            appendMessage(text, 'user');
            userInput.value = '';
            setUIState(true);
            typingIndicator.style.display = 'block';
